# UBSI API Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a private, unofficial REST API aggregating UBSI web services (studentv2, elearning, elibrary, ejournal, repository, news) with Redis caching, running on 127.0.0.1:8300.

**Architecture:** A lightweight FastAPI server with modular scrapers powered by Scrapling (impersonate="chrome"). Redis provides a two-tier caching strategy (fresh with 60s TTL, last-known-good with no TTL) for resilient upstream downtime tolerance. All endpoints return a standardized JSON envelope.

**Tech Stack:** Python 3.12+, FastAPI, Uvicorn, Scrapling, redis-py, pytest.

**Spec:** `docs/superpowers/specs/2026-09-25-ubsi-api-design.md`

## Global Constraints

- Binding: 127.0.0.1:8300 only — never 0.0.0.0.
- No API authentication headers required internally (localhost security boundary).
- Secrets: read from `.env` only (never logged, never committed, never returned in responses).
- Parser isolation: pure parsing functions `parse_*(html) -> dict` tested against snapshot fixtures without live network calls.
- Upstream stability: exponential backoff retries with reasonable sleep intervals between requests.

## Review Focus

1. Upstream down / HTML changed: API must return last-known-good data with `stale_since` timestamp rather than crashing with 500.
2. Login session expiry: auto re-login on redirect to `/login` without requiring client retry.
3. Elibrary extreme latency: 60s timeout with 3x retry before returning 502 or cached data.
4. Math captcha variations: regex for arithmetic questions must handle spacing and unexpected token ordering.
5. Incomplete upstream tables: absent table rows or columns must return empty lists or nulls, never indexing errors.

```python
# app/envelope.py
from datetime import datetime, timezone, timedelta
from typing import Any, Optional

WIB = timezone(timedelta(hours=7))

def make_envelope(
    data: Any,
    source: str,
    stale_since: Optional[str] = None
) -> dict[str, Any]:
    return {
        "ok": True,
        "data": data,
        "source": source,
        "fetched_at": datetime.now(WIB).isoformat(),
        "stale_since": stale_since,
    }

def error_envelope(module: str, kind: str, detail: str) -> dict[str, Any]:
    return {
        "ok": False,
        "error": {
            "module": module,
            "kind": kind,
            "detail": detail,
        },
        "fetched_at": datetime.now(WIB).isoformat(),
    }
```

```python
# app/cache.py
import hashlib
import json
from typing import Any, Optional
import redis.asyncio as aioredis

class CacheManager:
    def __init__(self, redis_url: str, ttl: int = 60):
        self.redis_url = redis_url
        self.ttl = ttl
        self._client: Optional[aioredis.Redis] = None

    async def get_client(self) -> aioredis.Redis:
        if self._client is None:
            self._client = aioredis.from_url(self.redis_url, decode_responses=True)
        return self._client

    def make_key(self, module: str, path: str, **params: Any) -> str:
        param_str = json.dumps(params, sort_keys=True)
        h = hashlib.sha256(f"{module}:{path}:{param_str}".encode()).hexdigest()[:16]
        return f"ubsi:{module}:{h}"

    async def get_fresh(self, key: str) -> Optional[dict]:
        client = await self.get_client()
        raw = await client.get(f"{key}:fresh")
        return json.loads(raw) if raw else None

    async def get_lgg(self, key: str) -> Optional[tuple[dict, str]]:
        client = await self.get_client()
        raw = await client.get(f"{key}:lgg")
        if not raw:
            return None
        data = json.loads(raw)
        stale_since = data.get("_saved_at", "unknown")
        return data.get("payload"), stale_since

    async def set(self, key: str, payload: Any) -> None:
        client = await self.get_client()
        import datetime
        now = datetime.datetime.now().isoformat()
        body = json.dumps(payload)
        lgg_body = json.dumps({"payload": payload, "_saved_at": now})
        await client.set(f"{key}:fresh", body, ex=self.ttl)
        await client.set(f"{key}:lgg", lgg_body)
```

```python
from contextlib import asynccontextmanager
from fastapi import FastAPI, status
from fastapi.responses import JSONResponse
from app.config import settings
from app.cache import CacheManager

cache = CacheManager(settings.REDIS_URL, ttl=settings.CACHE_TTL_SECONDS)

@asynccontextmanager
async def lifespan(app: FastAPI):
    yield
    client = await cache.get_client()
    await client.aclose()

app = FastAPI(
    title="UBSI API",
    description="Private Unofficial API Aggregator for UBSI Services",
    version="1.0.0",
    lifespan=lifespan
)

@app.get("/health", status_code=status.HTTP_200_OK)
async def health():
    redis_status = "down"
    try:
        client = await cache.get_client()
        pong = await client.ping()
        if pong:
            redis_status = "up"
    except Exception:
        redis_status = "down"
    return {"status": "ok", "redis": redis_status}
```

```python
# app/modules/studentv2.py
import re
import hashlib
from typing import Any, Optional
from scrapling.fetchers import FetcherSession
from scrapling.core import Adaptor

def parse_schedule(html: str) -> list[dict[str, Any]]:
    page = Adaptor(html)
    rows = page.css("table tr")
    results = []
    for row in rows[1:]:
        cols = [c.text.strip() for c in row.css("td")]
        if len(cols) >= 9:
            raw_dosen = cols[3].replace("\n", " ").strip()
            dosen_code = raw_dosen.split("/")[0].strip() if "/" in raw_dosen else raw_dosen
            item_id = hashlib.sha256(f"{cols[4]}:{cols[1]}:{cols[2]}".encode()).hexdigest()[:12]
            results.append({
                "id": item_id,
                "kode": cols[4],
                "name": cols[5],
                "day_of_week": cols[1],
                "jam": cols[2],
                "sks": int(cols[6]) if cols[6].isdigit() else 0,
                "kelompok_praktek": cols[7],
                "room": cols[8],
                "kode_dosen": dosen_code,
            })
    return results

def parse_grades(html: str) -> list[dict[str, Any]]:
    page = Adaptor(html)
    rows = page.css("table tr")
    results = []
    for row in rows[1:]:
        cols = [c.text.strip() for c in row.css("td")]
        if len(cols) >= 10:
            item_id = hashlib.sha256(cols[1].encode()).hexdigest()[:12]
            results.append({
                "id": item_id,
                "kode": cols[1],
                "name": cols[2],
                "sks": int(cols[3]) if cols[3].isdigit() else 0,
                "nilai_uts": cols[4],
                "nilai_uas": cols[5],
                "nilai_tugas": cols[6],
                "nilai_absen": cols[7],
                "total": cols[8],
                "grade": cols[9],
            })
    return results

def parse_news(html: str) -> list[dict[str, Any]]:
    page = Adaptor(html)
    rows = page.css("table tr")
    results = []
    for row in rows[1:]:
        cols = [c.text.strip() for c in row.css("td")]
        links = [a.attrib.get("href") for a in row.css("a") if a.attrib.get("href")]
        if len(cols) >= 4 and links:
            item_id = hashlib.sha256(cols[1].encode()).hexdigest()[:12]
            results.append({
                "id": item_id,
                "title": cols[1],
                "pdf_url": links[0],
                "date": cols[3],
            })
    return results

def parse_announcements(html: str) -> list[dict[str, Any]]:
    page = Adaptor(html)
    results = []
    links = page.css('a[href*="pengumuman_dalam_"]')
    for a in links:
        title = a.text.strip()
        url = a.attrib.get("href", "")
        if title and url:
            item_id = hashlib.sha256(url.encode()).hexdigest()[:12]
            results.append({
                "id": item_id,
                "title": title,
                "pdf_url": url,
            })
    return results
```
